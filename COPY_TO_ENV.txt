# ============================================================================
# LLM BACKEND SELECTION
# ============================================================================
# Which client the agent loop reasons with. Cortex is the default and is the

#   cortex = Snowflake Cortex Messages API (Claude)   [submission path]

# ============================================================================
# SNOWFLAKE CORTEX (Required when LLM_BACKEND=cortex)
# ============================================================================
# Auth is a Programmatic Access Token sent as a bearer token, NOT the
# SNOWFLAKE_PASSWORD above. A PAT will not authenticate until a network policy
# covering your IP exists on the account. Set an explicit expiry when you
# create it; the default is 15 days and the contest runs into September.
SNOWFLAKE_PAT=your_programmatic_access_token

# Model id is account- and region-specific. Confirm before first use with:
#   SHOW CORTEX BASE MODELS;
# Do not assume this default is correct for your account.

# Per-request timeout in seconds.
CORTEX_TIMEOUT_SECONDS=60

# Prompt caching. The agent resends a large stable prefix (system prompt plus
# the 35-tool schema) every turn, so leave this on: cached input reads at 0.15
# credits per million tokens against 1.80 uncached.

# Note: SNOWFLAKE_ACCOUNT above is reused for the Cortex host and must be the

# ============================================================================
# MLX LOCAL LLM (Optional - development backend, Apple Silicon only)
# ============================================================================
# Only consulted when LLM_BACKEND=mlx. Uses mlx-lm via an OpenAI-compatible
# server. Not a submission path: the contest rewards Cortex-native reasoning,
# and 35-tool dispatch is materially less reliable on a quantized local model.
# Start the server:
#   mlx_lm.server --model mlx-community/Qwen3-32B-4bit --port 8080
#
# Available models (cached via HuggingFace):
#   mlx-community/Qwen3-32B-4bit                   (main - general tasks)
#   mlx-community/QwQ-32B-4bit                      (reasoning - complex analysis)
#   mlx-community/Qwen2.5-Coder-14B-Instruct-4bit  (code/SQL - technical tasks)
#   mlx-community/Llama-3.2-3B-Instruct-4bit       (fast - simple queries)
